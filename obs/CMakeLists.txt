cmake_minimum_required(VERSION 3.20)
project(ytmusic-obs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
find_package(libobs CONFIG REQUIRED)
find_package(obs-frontend-api CONFIG REQUIRED)
find_package(Qt6 REQUIRED COMPONENTS Widgets Network Concurrent)
add_library(ytmusic-obs MODULE plugin.cpp)
target_link_libraries(ytmusic-obs PRIVATE OBS::libobs OBS::obs-frontend-api Qt6::Widgets Qt6::Network Qt6::Concurrent)
set_target_properties(ytmusic-obs PROPERTIES PREFIX "")
install(TARGETS ytmusic-obs LIBRARY DESTINATION lib/obs-plugins)
option(YTM_BUILD_SMOKE "Build headless native plugin smoke test" ON)
if(YTM_BUILD_SMOKE)
  find_package(PkgConfig REQUIRED)
  pkg_check_modules(WAYLAND REQUIRED IMPORTED_TARGET wayland-client)
  add_executable(ytmusic-obs-smoke smoke.cpp)
  target_link_libraries(ytmusic-obs-smoke PRIVATE OBS::libobs Qt6::Widgets Qt6::Network PkgConfig::WAYLAND)
  set_target_properties(ytmusic-obs-smoke PROPERTIES ENABLE_EXPORTS ON)
endif()
